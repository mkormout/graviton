## Replacement pattern (canonical)

```gdscript
# One-shot delay — replaces `Timer` nodes with `one_shot = true` and single-use timeouts
var _cooldown_left: float = 0.0

func _physics_process(delta: float) -> void:
    if _cooldown_left > 0.0:
        _cooldown_left -= delta
        if _cooldown_left <= 0.0:
            _cooldown_left = 0.0
            _on_cooldown_finished()

# "Start" the timer
_cooldown_left = WAIT_TIME

# "Stop" the timer
_cooldown_left = 0.0

# "is_stopped()" equivalent
func _is_running() -> bool:
    return _cooldown_left > 0.0
```

```gdscript
# Repeating timer (e.g. swarmer FireTimer with one_shot=false)
var _fire_interval: float = 0.8
var _fire_left: float = 0.0
var _fire_active: bool = false

func _physics_process(delta: float) -> void:
    if not _fire_active:
        return
    _fire_left -= delta
    if _fire_left <= 0.0:
        _fire_left += _fire_interval  # += preserves sub-frame drift (matches Timer behavior)
        _on_fire_timeout()

# "start()"
_fire_active = true
_fire_left = _fire_interval

# "stop()"
_fire_active = false
_fire_left = 0.0
```

```gdscript
# `await get_tree().create_timer(X).timeout` replacement inside a spawned entity.
# Option A (preferred when inside `_physics_process` / `_ready` on a Node with physics):
#   hoist the logic into a state machine with `_pending_left: float` and a `_pending_action` enum.
# Option B (acceptable when it's a single linear `_ready` sequence that must pause):
#   wait N physics frames using a pre-computed frame count:
#     var frames := int(ceil(delay / (1.0 / Engine.physics_ticks_per_second)))
#     for i in range(frames): await get_tree().physics_frame
#   This still avoids creating a SceneTreeTimer node and costs only N frame yields.
# Choose per site — see each task for the chosen option.
```
